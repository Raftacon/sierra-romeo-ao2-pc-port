# Rebuild the SDK's exact spdlog revision against the local MSVC STL. The
# release SDK's static archive was built with a newer STL than VS 17.9.
option(AOT_BUILD_HOST_SPDLOG "Build SDK logging dependency with the host toolchain" ON)
if(AOT_BUILD_HOST_SPDLOG AND WIN32)
    include(FetchContent)
    FetchContent_Declare(aot_spdlog_source
        URL https://github.com/gabime/spdlog/archive/79524ddd08a4ec981b7fea76afd08ee05f83755d.zip
        URL_HASH SHA256=2594162a28f06d12bf38ab4388c002156db9f8d5f7bcb0e1b4e60067dbad6c76
        # Archive timestamps may be outside the destination filesystem range (exFAT).
        DOWNLOAD_EXTRACT_TIMESTAMP FALSE
        SOURCE_SUBDIR aot-do-not-add-upstream-targets
    )
    FetchContent_MakeAvailable(aot_spdlog_source)
    add_library(aot_host_spdlog STATIC
        ${aot_spdlog_source_SOURCE_DIR}/src/spdlog.cpp
        ${aot_spdlog_source_SOURCE_DIR}/src/stdout_sinks.cpp
        ${aot_spdlog_source_SOURCE_DIR}/src/color_sinks.cpp
        ${aot_spdlog_source_SOURCE_DIR}/src/file_sinks.cpp
        ${aot_spdlog_source_SOURCE_DIR}/src/async.cpp
        ${aot_spdlog_source_SOURCE_DIR}/src/cfg.cpp
    )
    get_target_property(_spdlog_includes spdlog::spdlog INTERFACE_INCLUDE_DIRECTORIES)
    target_include_directories(aot_host_spdlog PRIVATE ${_spdlog_includes})
    target_compile_definitions(aot_host_spdlog PRIVATE SPDLOG_COMPILED_LIB SPDLOG_FMT_EXTERNAL)
    target_link_libraries(aot_host_spdlog PRIVATE fmt::fmt)
    set_target_properties(aot_host_spdlog PROPERTIES
        ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/host-libs")
    foreach(_cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(spdlog::spdlog PROPERTIES
            IMPORTED_LOCATION_${_cfg} "${CMAKE_BINARY_DIR}/host-libs/aot_host_spdlog.lib")
    endforeach()
    add_dependencies(spdlog::spdlog aot_host_spdlog)
endif()
